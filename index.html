// ESP32 C++ FIRMWARE IMPLEMENTATION
void IRAM_ATTR handleP1Press() {
  if (currentState == RUNNING_TEST && !p1Pressed) {
    p1PressTime = micros(); // 1us resolution
    p1Pressed = true;
  }
}

void setup() {
  pinMode(25, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(25), handleP1Press, FALLING);
}
